🎨 COLOR CUSTOMIZATION
════════════════════════════════════════════════════════════════════════════════

Apply colors based on POI type:

{
  "id": "poi-icons",
  "type": "symbol",
  "layout": {
    "icon-image": "restaurant-15"
  },
  "paint": {
    "icon-color": [
      "match",
      ["get", "class"],
      "restaurant", "#d97200",   // Orange
      "hotel", "#5d60be",        // Blue
      "shop", "#76a723",         // Green
      "#626262"                  // Gray
    ]
  }
}
